create database myproject1;
use myproject1;

create table doctors(
doctor_id	varchar(30),
doctor_name varchar(30),
specialty varchar(30),
hire_date date  not null,
rating int ,
employment_type varchar(30),
is_active varchar(30),
primary key(doctor_id));

create table patients(
patient_id varchar(30),
patient_name varchar(30),
age	tinyint not null,
gender	varchar(30) 
,city	varchar(30),
patient_type varchar(30)	
,preferred_time_slot	varchar(30) ,
registration_date date,
primary key(patient_id));


create table   rooms(  
room_id	varchar(30) primary key,
room_type varchar(30)  not null ,	
flor int ,
equipment_type varchar(30)
,capacity int ,	
last_maintenance_date date	
,is_available varchar(30));

CREATE TABLE appointments(
    appointment_id VARCHAR(30) PRIMARY KEY,
    patient_id VARCHAR(30),
    appointment_date DATE,
    doctor_id VARCHAR(30),
    service_type VARCHAR(30),
    priority VARCHAR(30),
    estimated_cost DOUBLE,
    booking_channel VARCHAR(30),
    FOREIGN KEY(patient_id) REFERENCES patients(patient_id),
    FOREIGN KEY(doctor_id) REFERENCES doctors(doctor_id)
);
CREATE TABLE treatments(
    treatment_id VARCHAR(30),
    appointment_id VARCHAR(30),
    doctor_id VARCHAR(30),
    room_id VARCHAR(30),
    actual_treatment_date DATE,
    treatment_status VARCHAR(30),
    treatment_attempt INT,
    treatment_duration_min INT,
    waiting_time_min INT,
    treatment_cost DOUBLE,
    FOREIGN KEY(appointment_id) REFERENCES appointments(appointment_id),
    PRIMARY KEY(treatment_id),
    FOREIGN KEY(doctor_id) REFERENCES doctors(doctor_id),
    FOREIGN KEY(room_id) REFERENCES rooms(room_id)
);

-- 10. Compare treatment performance over time based on appointment date and treatment status       
       
SELECT a.appointment_date,t.treatment_status, COUNT(t.treatment_id) AS total_treatments
FROM appointments a
JOIN treatments t ON a.appointment_id = t.appointment_id
GROUP BY a.appointment_date, t.treatment_status
ORDER BY a.appointment_date;
